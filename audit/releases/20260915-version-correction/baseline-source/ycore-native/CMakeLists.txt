cmake_minimum_required(VERSION 3.20)
project(ycore_native LANGUAGES C CXX)

add_library(ycore SHARED src/ycore.cpp src/ycore_build.cpp)
target_include_directories(ycore PUBLIC include)
target_compile_features(ycore PUBLIC cxx_std_17)
target_compile_options(ycore PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(ycore PRIVATE YCORE_BUILDING_LIBRARY=1)
set_target_properties(ycore PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

option(YCORE_BUILD_TESTS "Build host-side YCore ABI tests" ON)
if(YCORE_BUILD_TESTS)
    enable_testing()
    add_executable(ycore_test tests/ycore_test.cpp)
    target_link_libraries(ycore_test PRIVATE ycore)
    target_compile_features(ycore_test PRIVATE cxx_std_17)
    target_compile_options(ycore_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME ycore_test COMMAND ycore_test)
endif()
